                  selectedConstellationNode.artist.split(/(?:,\s*|&)/).map((artistName, index, arr) => {
                    const trimmedName = artistName.trim();
                    if (!trimmedName) return null;
                    return (
                      <span key={index}>
                        <button
                          onClick={() => {
                            setSelectedTrack({
                              id: "",
                              name: "",
                              artists: [{ id: `virt_${index}`, name: trimmedName }],
                              album: { images: [] },
                              preview_url: null,
                              external_urls: { spotify: "" }
                            } as any);
                            generateArtistDNA(trimmedName);
                          }}
                          className="text-emerald-400 hover:text-emerald-300 hover:underline font-bold transition-all cursor-pointer inline"
                          title={`Explore ${trimmedName}'s Music Map`}
                        >
                          {trimmedName}
                        </button>
                        {index < arr.length - 1 && <span className="text-slate-500 mx-1.5">&</span>}
                      </span>
                    );
                  })
                ) : (
                  "Unknown Artist"
                )}
              </p>

              {selectedConstellationNode.musicBrainzReleaseId && (
                <div className="flex items-center space-x-2 mt-1.5 bg-emerald-500/5 border border-emerald-500/10 rounded-lg px-2 py-0.5 max-w-fit">
                  <span className="text-[8px] font-extrabold text-emerald-400 font-mono uppercase tracking-wider">
                    Wikidata Mapped
                  </span>
                  <span className="h-1 w-1 bg-emerald-400 rounded-full" />
                  <a
                    href={`https://musicbrainz.org/release-group/${selectedConstellationNode.musicBrainzReleaseId}`}
                    target="_blank"
                    rel="noreferrer"
                    className="text-[9px] text-[#2dd4bf] hover:text-[#5eead4] font-mono flex items-center space-x-1 hover:underline cursor-pointer font-bold lowercase"
                    title={`View this release group on MusicBrainz`}
                  >
                    <span>mbid (release-group): {selectedConstellationNode.musicBrainzReleaseId.substring(0, 8)}...</span>
                    <span className="text-[8px]">↗</span>
                  </a>
                </div>
              )}
              
              <p className="text-xs text-slate-300 leading-normal font-sans font-light mt-2 bg-[#050505]/30 p-2.5 rounded-xl border border-white/5">
                {selectedConstellationNode.explanation}
              </p>
            </div>

            <div className="flex-shrink-0 flex sm:flex-col items-end gap-3 self-stretch sm:self-center justify-between sm:justify-center">
              <div className="text-right leading-none">
                <span className="text-[10px] text-slate-500 font-mono">SIMILARITY</span>
                <p className="text-lg font-bold font-mono text-emerald-400 mt-0.5">{selectedConstellationNode.similarityScore}%</p>
              </div>
              
              <button
                onClick={() => traverseToRecommendation(selectedConstellationNode)}
                className="flex items-center space-x-1 bg-emerald-500 hover:bg-emerald-400 text-black font-semibold text-[10px] uppercase tracking-wider py-2.5 px-4 rounded-full transition-all shadow-lg shadow-[#10b981]/15 cursor-pointer"
              >
                <span>Model Node</span>
                <ChevronRight className="w-3.5 h-3.5 stroke-[2.5px]" />
              </button>
            </div>
          </div>
        ) : (
          <div className="text-center p-6 border border-white/5 rounded-2xl border-dashed">
            <p className="text-xs text-slate-500 font-mono uppercase tracking-tight">Highlight a constellation star node above to display evaluation relationship data.</p>
          </div>
        )}

      </div>
    );
  };
